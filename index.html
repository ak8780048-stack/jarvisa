import os import json import base64 import asyncio import traceback from fastapi import FastAPI, WebSocket, WebSocketDisconnect from fastapi.middleware.cors import CORSMiddleware from google import genai from google.genai import types app = FastAPI(title="Jarvis Multimodal Core") app.add_middleware( CORSMiddleware, allow_origins=["*"], allow_methods=["*"], allow_headers=["*"], ) # Initialize Gemini Client (v1alpha required for Multimodal Live API) client = genai.Client(http_options={"api_version": "v1alpha"}) LIVE_MODEL = "gemini-2.0-flash-exp" # Supports bidirectional audio/video streaming IMAGE_MODEL = "imagen-3.0-generate-002" # Tool Definition: Image Generation generate_image_tool = types.Tool( function_declarations=[ types.FunctionDeclaration( name="generate_image", description="Generate a high-quality image based on a detailed visual prompt when the user asks to create, draw, or visualize something.", parameters=types.Schema( type=types.Type.OBJECT, properties={ "prompt": types.Schema( type=types.Type.STRING, description="Detailed, descriptive prompt for image generation.", ), "aspect_ratio": types.Schema( type=types.Type.STRING, description="Aspect ratio like '1:1', '16:9', or '9:16'. Default is '1:1'.", ), }, required=["prompt"], ), ) ] ) # Live Session Configuration (Emotive TTS + System Persona) LIVE_CONFIG = types.LiveConnectConfig( response_modalities=[types.LiveModality.AUDIO], speech_config=types.SpeechConfig( voice_config=types.VoiceConfig( prebuilt_voice_config=types.PrebuiltVoiceConfig( voice_name="Puck" # Options: Puck, Charon, Kore, Fenrir, Aoede ) ) ), system_instruction=types.Content( parts=[ types.Part.from_text( text=( "You are Jarvis, an advanced, witty, and hyper-capable multimodal AI assistant. " "You can see through the user's camera feed and hear their voice in real time. " "Respond naturally with appropriate emotional inflection, keeping answers concise " "unless deep technical detail is requested. Use the `generate_image` tool whenever " "the user asks you to create or show an image." ) ) ] ), tools=[generate_image_tool], ) async def execute_image_generation(prompt: str, aspect_ratio: str = "1:1") -> str: """Generates an image using Imagen 3 and returns a Base64 encoded JPEG string.""" loop = asyncio.get_running_loop() def _call_imagen(): result = client.models.generate_images( model=IMAGE_MODEL, prompt=prompt, config=types.GenerateImagesConfig( number_of_images=1, output_mime_type="image/jpeg", aspect_ratio=aspect_ratio, person_generation="ALLOW_ADULT", ), ) for generated_image in result.generated_images: return base64.b64encode(generated_image.image.image_bytes).decode("utf-8") return "" return await loop.run_in_executor(None, _call_imagen) @app.websocket("/ws/jarvis") async def jarvis_websocket_endpoint(websocket: WebSocket): await websocket.accept() print("[Jarvis] Client connected.") try: async with client.aio.live.connect(model=LIVE_MODEL, config=LIVE_CONFIG) as session: async def receive_from_client(): """Receives Audio (PCM), Video Frames (JPEG), and Text from Frontend.""" try: while True: raw_data = await websocket.receive_text() payload = json.loads(raw_data) msg_type = payload.get("type") if msg_type == "audio": # Expects Base64 encoded 16kHz, 1-channel, 16-bit PCM await session.send( input={"data": payload["data"], "mime_type": "audio/pcm"}, end_of_turn=False, ) elif msg_type == "video": # Expects Base64 encoded JPEG frame (throttled to ~1 FPS on client) await session.send( input={"data": payload["data"], "mime_type": "image/jpeg"}, end_of_turn=False, ) elif msg_type == "text": await session.send( input=payload["data"], end_of_turn=True, ) except WebSocketDisconnect: print("[Jarvis] Client disconnected from upstream.") except Exception as e: print(f"[Upstream Error]: {e}") async def send_to_client(): """Streams Gemini Audio, Transcripts, and Generated Images back to Frontend.""" try: async for response in session.receive(): server_content = response.server_content if server_content is not None: model_turn = server_content.model_turn if model_turn is not None: for part in model_turn.parts: # 1. Stream Emotive TTS Audio (24kHz PCM) if part.inline_data: audio_b64 = base64.b64encode(part.inline_data.data).decode("utf-8") await websocket.send_json({ "type": "audio", "data": audio_b64 }) # 2. Stream Text Transcript (if available) if part.text: await websocket.send_json({ "type": "text", "data": part.text }) # Handle user interruption (barge-in) if server_content.interrupted: await websocket.send_json({"type": "interrupted"}) # 3. Handle Tool Calls (Multimodal Output Generation) tool_call = response.tool_call if tool_call is not None: for call in tool_call.function_calls: if call.name == "generate_image": args = call.args prompt = args.get("prompt", "") aspect_ratio = args.get("aspect_ratio", "1:1") await websocket.send_json({ "type": "status", "data": f"Generating visual: {prompt}..." }) img_b64 = await execute_image_generation(prompt, aspect_ratio) # Send generated image directly to UI if img_b64: await websocket.send_json({ "type": "generated_image", "data": img_b64, "prompt": prompt }) # Acknowledge tool execution back to Gemini so it can speak about it await session.send( input=types.LiveClientToolResponse( function_responses=[ types.FunctionResponse( name="generate_image", id=call.id, response={"result": "Image generated and displayed to user."} ) ] ) ) except Exception as e: print(f"[Downstream Error]: {e}") traceback.print_exc() # Run upstream and downstream loops concurrently await asyncio.gather(receive_from_client(), send_to_client()) except Exception as e: print(f"[Session Error]: {e}") finally: print("[Jarvis] Session closed.") if __name__ == "__main__": import uvicorn uvicorn.run(app, host="0.0.0.0", port=8000)
